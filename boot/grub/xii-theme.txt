# ────────────────────────────────────────────────
# Boot Menu Box
# ────────────────────────────────────────────────
# + item_color          : color of normal menu items
# + selected_item_color : color of highlighted item
# + item_height         : height in pixels per item
# + item_padding        : internal padding
# + item_spacing        : gap between items

+ boot_menu {
    left   = 20%
    top    = 28%
    width  = 60%
    height = 45%

    icon_height             = 24
}
